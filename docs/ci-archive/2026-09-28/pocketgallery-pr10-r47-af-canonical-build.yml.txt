name: PocketGallery PR10 R4.7 OKF A-F Canonical Build

on:
  pull_request:
    branches:
      - feature/phone-pilot-r4-chat-first
    paths:
      - '.github/workflows/pocketgallery-pr10-r47-af-canonical-build.yml'
  workflow_dispatch:

permissions:
  contents: write

concurrency:
  group: pocketgallery-pr10-r47-af-canonical-build
  cancel-in-progress: false

jobs:
  canonical-af-apk:
    runs-on: ubuntu-latest
    timeout-minutes: 90
    env:
      TARGET_BRANCH: codex/r47-okf-af-benchmark
      CANONICAL_SIGNER_SHA256: 81af4a5ef94c236774f0e193b2a4b248805b36c14cc36e2a56df8e451a712541
      EXPECTED_VERSION_CODE: '2024'
      EXPECTED_VERSION_NAME: '0.5.0'

    steps:
      - name: Checkout PR10 signing carrier
        uses: actions/checkout@v4

      - name: Verify PR10 execution scope
        shell: bash
        run: |
          set -euo pipefail
          echo "EVENT=$GITHUB_EVENT_NAME"
          echo "REF=$GITHUB_REF"
          if [[ "$GITHUB_EVENT_NAME" = "pull_request" ]]; then
            test "$GITHUB_REF" = "refs/pull/10/merge"
          fi

      - name: Restore PR10 canonical R3 signer
        id: signer-cache
        uses: actions/cache@v5
        with:
          path: ~/.android/pocketgallery-r3-signing
          key: pocketgallery-r3-signing-v1

      - name: Verify canonical signer and export build environment
        shell: bash
        run: |
          set -euo pipefail
          test "${{ steps.signer-cache.outputs.cache-hit }}" = "true"
          SIGN_DIR="$HOME/.android/pocketgallery-r3-signing"
          test -s "$SIGN_DIR/r3.keystore"
          test -s "$SIGN_DIR/password"
          PASS="$(<"$SIGN_DIR/password")"
          echo "::add-mask::$PASS"
          keytool -exportcert \
            -keystore "$SIGN_DIR/r3.keystore" \
            -storepass "$PASS" \
            -alias pocketgallery-r3 \
            -file "$RUNNER_TEMP/pr10-af-canonical.der" >/dev/null
          CERT_SHA="$(sha256sum "$RUNNER_TEMP/pr10-af-canonical.der" | awk '{print $1}')"
          echo "RESTORED_SIGNER_SHA256=$CERT_SHA"
          test "$CERT_SHA" = "$CANONICAL_SIGNER_SHA256"
          {
            echo "POCKETGALLERY_SIGNING_KEYSTORE=$SIGN_DIR/r3.keystore"
            echo "POCKETGALLERY_SIGNING_STORE_PASSWORD=$PASS"
            echo "POCKETGALLERY_SIGNING_KEY_PASSWORD=$PASS"
            echo "POCKETGALLERY_SIGNING_KEY_ALIAS=pocketgallery-r3"
          } >> "$GITHUB_ENV"

      - name: Resolve exact latest A-F target SHA
        shell: bash
        run: |
          set -euo pipefail
          TARGET_SHA="$(git ls-remote --exit-code origin "refs/heads/$TARGET_BRANCH" | awk '{print $1}')"
          test -n "$TARGET_SHA"
          echo "TARGET_SHA=$TARGET_SHA" >> "$GITHUB_ENV"
          echo "LATEST_AF_SHA=$TARGET_SHA"

      - name: Checkout exact A-F source
        uses: actions/checkout@v4
        with:
          ref: ${{ env.TARGET_SHA }}
          fetch-depth: 1
          persist-credentials: true

      - name: Verify exact source identity
        shell: bash
        run: |
          set -euo pipefail
          test "$(git rev-parse HEAD)" = "$TARGET_SHA"
          git diff --exit-code
          echo "SOURCE_SHA=$(git rev-parse HEAD)"
          echo "SOURCE_TREE=$(git rev-parse 'HEAD^{tree}')"

      - uses: subosito/flutter-action@v2
        with:
          channel: stable
          cache: true

      - name: Resolve Flutter dependencies
        working-directory: pilot/flutter_phone_loop
        run: flutter pub get

      - name: Analyze
        working-directory: pilot/flutter_phone_loop
        run: flutter analyze

      - name: A-F focused contracts
        working-directory: pilot/flutter_phone_loop
        run: flutter test test/r47_okf_af_benchmark_test.dart test/r47_okf_af_ui_test.dart

      - name: Full Flutter regression suite
        working-directory: pilot/flutter_phone_loop
        run: flutter test

      - name: Bootstrap Android scaffold
        working-directory: pilot/flutter_phone_loop
        run: bash scripts/bootstrap_android.sh

      - name: Build canonical arm64 A-F APK
        working-directory: pilot/flutter_phone_loop
        shell: bash
        run: |
          set -euo pipefail
          flutter build apk --debug \
            --build-number=24 \
            --dart-define=POCKETGALLERY_SOURCE_COMMIT="$TARGET_SHA" \
            --target-platform android-arm64 \
            --split-per-abi
          SRC=build/app/outputs/flutter-apk/app-arm64-v8a-debug.apk
          OUT=build/app/outputs/flutter-apk/PocketGallery-R47-OKF-AF-Benchmark-arm64.apk
          test -s "$SRC"
          cp "$SRC" "$OUT"

      - name: Verify package version ABI signer and hash
        working-directory: pilot/flutter_phone_loop
        shell: bash
        run: |
          set -euo pipefail
          APK=build/app/outputs/flutter-apk/PocketGallery-R47-OKF-AF-Benchmark-arm64.apk
          ABIS="$(unzip -Z1 "$APK" | grep -E '^lib/[^/]+/.*\.so$' | sed -E 's#^lib/([^/]+)/.*#\1#' | sort -u | paste -sd, -)"
          echo "APK_ABIS=$ABIS"
          test "$ABIS" = "arm64-v8a"

          AAPT="$(find "$ANDROID_HOME/build-tools" -type f -name aapt | sort -V | tail -1)"
          BADGING="$("$AAPT" dump badging "$APK")"
          PACKAGE="$(printf '%s\n' "$BADGING" | sed -n "s/^package: name='\([^']*\)'.*/\1/p" | head -1)"
          VERSION_CODE="$(printf '%s\n' "$BADGING" | sed -n "s/^package:.*versionCode='\([^']*\)'.*/\1/p" | head -1)"
          VERSION_NAME="$(printf '%s\n' "$BADGING" | sed -n "s/^package:.*versionName='\([^']*\)'.*/\1/p" | head -1)"
          echo "APK_PACKAGE=$PACKAGE"
          echo "APK_VERSION_CODE=$VERSION_CODE"
          echo "APK_VERSION_NAME=$VERSION_NAME"
          test "$PACKAGE" = "com.qujindai.pocketgallery_phone_pilot.r3"
          test "$VERSION_CODE" = "$EXPECTED_VERSION_CODE"
          test "$VERSION_NAME" = "$EXPECTED_VERSION_NAME"

          APKSIGNER="$(find "$ANDROID_HOME/build-tools" -type f -name apksigner | sort -V | tail -1)"
          "$APKSIGNER" verify --verbose --print-certs "$APK" 2>&1 | tee "$RUNNER_TEMP/r47-af-apksigner.txt"
          APK_CERT="$(grep -i 'certificate.*sha-256.*digest' "$RUNNER_TEMP/r47-af-apksigner.txt" | head -1 | sed -E 's/.*[Dd]igest:[[:space:]]*//' | tr -d '[:space:]:' | tr '[:upper:]' '[:lower:]')"
          echo "APK_SIGNER_SHA256=$APK_CERT"
          test "$APK_CERT" = "$CANONICAL_SIGNER_SHA256"

          APK_SHA256="$(sha256sum "$APK" | awk '{print $1}')"
          printf '%s  %s\n' "$APK_SHA256" "$(basename "$APK")" > "$APK.sha256"
          {
            echo "SOURCE_SHA=$TARGET_SHA"
            echo "SOURCE_TREE=$(git rev-parse 'HEAD^{tree}')"
            echo "APK_PACKAGE=$PACKAGE"
            echo "APK_VERSION_CODE=$VERSION_CODE"
            echo "APK_VERSION_NAME=$VERSION_NAME"
            echo "APK_ABIS=$ABIS"
            echo "APK_SIGNER_SHA256=$APK_CERT"
            echo "APK_SHA256=$APK_SHA256"
          } | tee build/app/outputs/flutter-apk/R47_OKF_AF_SIGNED_BUILD_PROOF.txt

      - name: Upload canonical A-F APK artifact
        uses: actions/upload-artifact@v4
        with:
          name: PocketGallery-R47-OKF-AF-Benchmark-Canonical-arm64
          path: |
            pilot/flutter_phone_loop/build/app/outputs/flutter-apk/PocketGallery-R47-OKF-AF-Benchmark-arm64.apk
            pilot/flutter_phone_loop/build/app/outputs/flutter-apk/PocketGallery-R47-OKF-AF-Benchmark-arm64.apk.sha256
            pilot/flutter_phone_loop/build/app/outputs/flutter-apk/R47_OKF_AF_SIGNED_BUILD_PROOF.txt
          if-no-files-found: error
          retention-days: 14

      - name: Publish canonical A-F prerelease
        env:
          GH_TOKEN: ${{ github.token }}
        working-directory: pilot/flutter_phone_loop
        shell: bash
        run: |
          set -euo pipefail
          OUT=build/app/outputs/flutter-apk
          TAG="r47-okf-af-canonical-${GITHUB_RUN_ID}"
          gh release create "$TAG" \
            "$OUT/PocketGallery-R47-OKF-AF-Benchmark-arm64.apk#PocketGallery-R47-OKF-AF-Benchmark-arm64.apk" \
            "$OUT/PocketGallery-R47-OKF-AF-Benchmark-arm64.apk.sha256#PocketGallery-R47-OKF-AF-Benchmark-arm64.apk.sha256" \
            "$OUT/R47_OKF_AF_SIGNED_BUILD_PROOF.txt#R47_OKF_AF_SIGNED_BUILD_PROOF.txt" \
            --target "$TARGET_SHA" \
            --title "PocketGallery R4.7 OKF A-F canonical ${GITHUB_RUN_ID}" \
            --notes "Canonical R3-signed arm64 OKF A-F causal benchmark. Source ${TARGET_SHA}." \
            --prerelease
