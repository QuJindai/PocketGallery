  workflow_dispatch:

permissions:
  contents: write

concurrency:
  group: r47-okf-lint-fix
  cancel-in-progress: false

jobs:
  fix-and-verify:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
        with:
          ref: codex/r47-okf-gallery
          fetch-depth: 0

      - name: Apply analyzer-proven lint-only edits
        shell: bash
        run: |
          set -euo pipefail
          python3 - <<'PY'
          from pathlib import Path
          import re

          engine = Path('pilot/flutter_phone_loop/lib/okf/okf_experiment_engine.dart')
          text = engine.read_text()

          replacements = {
              'required LineageStore store,': 'required super.store,',
              'required LexicalFtsStore lexicalStore,': 'required super.lexicalStore,',
              'required RepresentationBuilder representationBuilder,': 'required super.representationBuilder,',
          }
          for old, new in replacements.items():
              count = text.count(old)
              if count != 1:
                  raise SystemExit(f'{old!r} count={count}; expected 1')
              text = text.replace(old, new)

          pattern = re.compile(
              r"\}\) : super\(\s*"
              r"store: store,\s*"
              r"lexicalStore: lexicalStore,\s*"
              r"representationBuilder: representationBuilder,\s*"
              r"clock: now,\s*"
              r"\);"
          )
          text, count = pattern.subn('}) : super(clock: now);', text)
          if count != 1:
              raise SystemExit(f'super initializer replacement count={count}; expected 1')

          unused_engine_imports = [
              "import '../experiments/representation_builder.dart';\n",
              "import '../lineage/lineage_store.dart';\n",
              "import '../services/lexical_fts_store.dart';\n",
          ]
          for unused in unused_engine_imports:
              count = text.count(unused)
              if count != 1:
                  raise SystemExit(f'engine import {unused!r} count={count}; expected 1')
              text = text.replace(unused, '')
          engine.write_text(text)

          test_file = Path('pilot/flutter_phone_loop/test/r47_okf_shadow_lab_test.dart')
          test_text = test_file.read_text()
          unused_test_import = "import 'package:pocketgallery_phone_pilot/okf/okf_models.dart';\n"
          count = test_text.count(unused_test_import)
          if count != 1:
              raise SystemExit(f'test import count={count}; expected 1')
          test_file.write_text(test_text.replace(unused_test_import, ''))
          PY
          git diff --check

      - uses: subosito/flutter-action@v2
        with:
          channel: stable
          cache: true

      - name: Resolve dependencies
        working-directory: pilot/flutter_phone_loop
        run: flutter pub get

      - name: Format only touched R4.7 files
        working-directory: pilot/flutter_phone_loop
        run: |

          git diff --check

      - name: Analyze full Flutter package

          mapfile -t TESTS < <(find test -maxdepth 1 -type f -name 'r47*.dart' -print | sort)
          test "${#TESTS[@]}" -gt 0
          flutter test "${TESTS[@]}"

      - name: Commit verified analyzer-only fix
        shell: bash
        run: |
          set -euo pipefail

            pilot/flutter_phone_loop/test/r47_okf_shadow_lab_test.dart
